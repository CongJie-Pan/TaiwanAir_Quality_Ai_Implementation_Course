## 1.5 數據前處理

### 1.5.1 特徵工程

```python
# 季節標籤
df['season'] = df['month'].map({
    12: '冬季', 1: '冬季', 2: '冬季',
    3: '春季', 4: '春季', 5: '春季',
    6: '夏季', 7: '夏季', 8: '夏季',
    9: '秋季', 10: '秋季', 11: '秋季'
})

# AQI 等級標籤
def aqi_level(aqi):
    if aqi <= 50: return '良好'
    elif aqi <= 100: return '普通'
    else: return '對敏感族群不健康'

df['aqi_level'] = df['aqi'].apply(aqi_level)

# 風速等級（對應期中交叉表）
df['wind_level'] = pd.cut(df['windspeed'],
    bins=[0, 1.5, 3.4, float('inf')],
    labels=['無風', '輕風', '微風以上'])
```
